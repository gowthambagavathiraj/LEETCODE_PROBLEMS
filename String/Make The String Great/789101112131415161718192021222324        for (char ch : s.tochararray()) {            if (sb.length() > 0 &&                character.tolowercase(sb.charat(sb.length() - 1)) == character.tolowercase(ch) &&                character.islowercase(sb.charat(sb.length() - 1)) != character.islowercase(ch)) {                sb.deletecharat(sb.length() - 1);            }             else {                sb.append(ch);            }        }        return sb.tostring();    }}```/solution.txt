        for (char ch : s.toCharArray()) {

            if (sb.length() > 0 &&
                Character.toLowerCase(sb.charAt(sb.length() - 1)) == Character.toLowerCase(ch) &&
                Character.isLowerCase(sb.charAt(sb.length() - 1)) != Character.isLowerCase(ch)) {

                sb.deleteCharAt(sb.length() - 1);
            } 
            else {
                sb.append(ch);
            }
        }

        return sb.toString();
    }
}
